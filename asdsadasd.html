Conexión con Google Sheets (Excel)

Al guardar esta URL, podrás compartir el enlace con tus operarios y sus reportes viajarán directo a tu Excel.

Código para Google Apps Script (Lectura + Escritura): 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["ID", "Fecha", "Zona", "Equipo", "Tipo", "Descripcion", "Operario", "Estado"]);
  }
  sheet.appendRow([data.id, data.fecha, data.zona, data.equipo, data.tipo, data.descripcion, data.operario, data.estado]);
  return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var result = [];
  for (var i = 1; i < rows.length; i++) {
    result.push({
      id: rows[i][0], fecha: rows[i][1], zona: rows[i][2], equipo: rows[i][3],
      tipo: rows[i][4], descripcion: rows[i][5], operario: rows[i][6], estado: rows[i][7]
    });
  }
  return ContentService.createTextOutput(JSON.stringify(result)).setMimeType(ContentService.MimeType.JSON);
}
